function validarCompatibilidad(jsonCrudo) {
    const resultado = {
        esValido: false,
        errores: [],
        advertencias: [],
        estadisticas: {}
    };
    try {
        if (!jsonCrudo || typeof jsonCrudo !== 'object') {
            resultado.errores.push('Entrada no es un objeto');
            return resultado;
        }
        if (!jsonCrudo.documento) {
            resultado.errores.push('Campo "documento" faltante');
        }
        if (!Array.isArray(jsonCrudo.contenido)) {
            resultado.errores.push('Campo "contenido" no es un array');
        } else {
            resultado.estadisticas.elementos = jsonCrudo.contenido.length;
        }
        resultado.esValido = resultado.errores.length === 0;
    } catch (error) {
        resultado.errores.push(`Error en validaciÃ³n: ${error.message}`);
    }
    return resultado;
}
